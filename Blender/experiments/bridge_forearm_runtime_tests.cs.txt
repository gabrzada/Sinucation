using System.Collections;
using System.Reflection;
using FriendSlop.Core;
using FriendSlop.Presentation;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.TestTools;

namespace FriendSlop.Tests
{
    public class BridgeForearmRuntimeTests
    {
        [UnityTest]
        public IEnumerator RebuildAndDestructionReleaseOwnedMeshesWithoutDestroyingSharedModel()
        {
            var flags=BindingFlags.Instance|BindingFlags.NonPublic;
            foreach(var shape in new[] {Archetype.Square,Archetype.Circle,Archetype.Triangle})
            {
                var root=new GameObject("BridgeForearmRuntime");Mesh finalMesh=null,original=null;
                try
                {
                    var avatar=AvatarVisual.Create(root.transform,new PlayerProfile("QA",shape,Color.cyan));
                    yield return null;
                    Mesh Bend()
                    {
                        var elbow=(Transform)typeof(AvatarVisual).GetField("_elbowL",flags).GetValue(avatar);
                        var filter=elbow.Find("Forearm").GetComponent<MeshFilter>();original=filter.sharedMesh;
                        avatar.LeftHand.rotation=Quaternion.AngleAxis(45f,elbow.right)*elbow.rotation;
                        typeof(AvatarVisual).GetMethod("UpdateBridgeForearm",flags).Invoke(avatar,new object[]{true});
                        Assert.That(filter.sharedMesh,Is.Not.SameAs(original));return filter.sharedMesh;
                    }
                    var old=Bend();avatar.Apply(new PlayerProfile("QA",shape,Color.red));
                    yield return null;yield return null;
                    Assert.That(old==null,Is.True,"appearance changes must release the first owned mesh: "+shape);
                    Assert.That(original!=null,Is.True,"shared model must survive an appearance change");
                    finalMesh=Bend();
                }
                finally {Object.Destroy(root);}
                yield return null;yield return null;
                Assert.That(finalMesh==null,Is.True,"runtime destruction must release the owned mesh: "+shape);
                Assert.That(original!=null,Is.True,"runtime destruction must preserve the shared model");
            }
        }
    }
}
