using System.IO;
using FriendSlop.Net;
using NUnit.Framework;
using UnityEngine;

namespace FriendSlop.Tests
{
    public class CueStrokePresentationTests
    {
        static CueStrokePresentation Sample() => new CueStrokePresentation {
            Sequence = 17, Shooter = 2, HasStartTime = true, StartedAt = 100.25, Tip = new Vector3(.1f, .9f, -.3f),
            Direction = Vector3.forward, Bridge = new Vector3(.1f, .9f, .5f),
            Length = 1.3f, ContactTravel = .24f, ExtraTravel = .12f
        };

        static byte[] Encode(CueStrokePresentation value)
        {
            using (var ms = new MemoryStream()) using (var w = new BinaryWriter(ms)) { value.Write(w); return ms.ToArray(); }
        }
        static bool Decode(byte[] bytes, out CueStrokePresentation value)
        {
            using (var ms = new MemoryStream(bytes)) using (var r = new BinaryReader(ms)) return CueStrokePresentation.TryRead(r, out value);
        }
        [Test]
        public void PreservesEveryStrokeParameter()
        {
            var sent = Sample(); Assert.That(Decode(Encode(sent), out var got), Is.True);
            Assert.That(got.StartedAt, Is.EqualTo(sent.StartedAt)); Assert.That(got.HasStartTime, Is.True);
            Assert.That(got.Valid, Is.True); Assert.That(got.Sequence, Is.EqualTo(sent.Sequence));
            Assert.That(got.Shooter, Is.EqualTo(sent.Shooter)); Assert.That(got.Tip, Is.EqualTo(sent.Tip));
            Assert.That(got.Direction, Is.EqualTo(sent.Direction)); Assert.That(got.Bridge, Is.EqualTo(sent.Bridge));
            Assert.That(got.Length, Is.EqualTo(sent.Length)); Assert.That(got.ContactTravel, Is.EqualTo(sent.ContactTravel));
            Assert.That(got.ExtraTravel, Is.EqualTo(sent.ExtraTravel));
        }
        [Test]
        public void RejectsUnknownVersionAndIncompletePacket()
        {
            var bytes = Encode(Sample()); bytes[0] = 3;
            Assert.That(Decode(bytes, out _), Is.False);
            for (int length = 0; length < bytes.Length; length++) Assert.That(Decode(new byte[length], out _), Is.False);
        }
        [Test]
        public void RejectsInvalidGeometryAndNonfiniteValues()
        {
            var value = Sample(); value.ExtraTravel = float.NaN; Assert.That(Decode(Encode(value), out _), Is.False);
            value = Sample(); value.Direction = Vector3.zero; Assert.That(Decode(Encode(value), out _), Is.False);
            value = Sample(); value.ContactTravel = 10f; Assert.That(Decode(Encode(value), out _), Is.False);
            value = Sample(); value.Tip.x = float.PositiveInfinity; Assert.That(Decode(Encode(value), out _), Is.False);
        }
    }
}
