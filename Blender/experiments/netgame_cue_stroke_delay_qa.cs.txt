namespace FriendSlop.Net
{
    public sealed partial class NetGame
    {
        CueStrokePresentation _latestCueStroke;
        FriendSlop.Match.MatchDirector _cueStrokeDirector;

        public bool TryGetCueStroke(int sequence, int shooter, out CueStrokePresentation stroke)
        {
            stroke = _latestCueStroke;
            return stroke.Valid && stroke.Sequence == sequence && stroke.Shooter == shooter && NetworkManager != null && stroke.ActiveAt(NetworkManager.ServerTime.Time);
        }

        public bool PublishCueStroke(CueStrokePresentation stroke)
        {
            if (!ReferenceEquals(_cueStrokeDirector, _director)) { _latestCueStroke = default; _cueStrokeDirector = _director; }
            if (!IsServer || _director == null || !stroke.IsValid ||
                stroke.Sequence != _director.ShotSequence || stroke.Shooter != _director.CurrentPlayerId ||
                (_latestCueStroke.Valid && stroke.Sequence <= _latestCueStroke.Sequence)) return false;
            stroke.Valid = true;
            _latestCueStroke = stroke;
            EmitEvent(NetEventType.CueStroke, stroke.Write);
            return true;
        }

        void ReceiveCueStroke(System.IO.BinaryReader reader)
        {
            if (!CueStrokePresentation.TryRead(reader, out var stroke)) return;
            float delay = 0f;
            var args = System.Environment.GetCommandLineArgs();
            for (int i = 0; i < args.Length - 1; i++)
                if (args[i] == "-fsCueDelayMs" && !IsServer && int.TryParse(args[i+1], out int ms)) delay = ms / 1000f;
            if (delay > 0f) StartCoroutine(DelayedCueStroke(stroke, delay));
            else AcceptCueStroke(stroke);
        }
        System.Collections.IEnumerator DelayedCueStroke(CueStrokePresentation stroke, float delay)
        {
            yield return new UnityEngine.WaitForSecondsRealtime(delay);
            AcceptCueStroke(stroke);
        }
        void AcceptCueStroke(CueStrokePresentation stroke)
        {
            double now = NetworkManager.ServerTime.Time;
            UnityEngine.Debug.Log("[CueStrokeArrival] sequence=" + stroke.Sequence + " start=" + stroke.StartedAt.ToString("F5") + " age=" + stroke.Elapsed(now).ToString("F5") + " active=" + stroke.ActiveAt(now));
            if (!_latestCueStroke.Valid || (stroke.HasStartTime ? stroke.StartedAt > _latestCueStroke.StartedAt : stroke.Sequence > _latestCueStroke.Sequence))
                _latestCueStroke = stroke;
        }
    }
}
