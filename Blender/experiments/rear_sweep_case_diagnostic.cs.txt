using FriendSlop.Billiards; using FriendSlop.Core; using FriendSlop.Presentation; using NUnit.Framework; using UnityEngine; using System.Reflection;
namespace FriendSlop.Tests { public class WristMotionDiagnostic {
[TestCase(Archetype.Square, .2f, .25f)] [TestCase(Archetype.Circle, .2f, .25f)] [TestCase(Archetype.Triangle, .2f, .25f)] public void Scan(Archetype shape, float offset, float mix) {
 Random.InitState(2026); var settings=new BilliardsPhysicsSettings();var table=new TableGeometry(settings);var holder=new GameObject("WristMotion");var avatar=AvatarVisual.Create(holder.transform,new PlayerProfile("QA",shape,Color.cyan));var flags=BindingFlags.Instance|BindingFlags.NonPublic;var elbow=(Transform)typeof(AvatarVisual).GetField("_elbowR",flags).GetValue(avatar); var shoulder=(Transform)typeof(AvatarVisual).GetField("_shoulderR",flags).GetValue(avatar);
 float worst=0f, sum=0f, worstStep=0f, worstAlign=1f; int count=0; try {for(int x=-1;x<=1;x++)for(int z=-1;z<=1;z++)for(int deg=0;deg<360;deg+=45){var ball=new Vector2(x*table.HalfLength*.7f,z*table.HalfWidth*.7f);float a=deg*Mathf.Deg2Rad;var d=new Vector2(Mathf.Sin(a),Mathf.Cos(a));var stand=ShotMath.StandBehind(ball,d,table);holder.transform.SetPositionAndRotation(new Vector3(stand.x,0,stand.y),Quaternion.Euler(0,ShotMath.WorldYaw(d),0));avatar.ResetMotionForStance();var flat=new Vector3(d.x,0,d.y);var tilt=Quaternion.AngleAxis(ShotMath.CueTilt(ball,d,table,settings.BallRadius),Vector3.Cross(Vector3.up,flat))*flat;var world=TableView.ToWorld(ball,settings.BallRadius);var tip0=world-tilt*(settings.BallRadius+.01f);var bridge=world-flat*.18f;bridge.y=TableView.TopY+.035f;float min=1,maxStep=0,maxCueError=0; Quaternion previous=Quaternion.identity; Vector3 elbowAnchor=Vector3.zero, elbowPrevious=Vector3.zero; float elbowTravel=0, elbowStep=0;
 for(int i=0;i<120;i++){float pull=i<60?0:(i-60)*.3f/59;float visual=avatar.VisualCuePull(tip0,tilt,pull);float length=Mathf.Clamp(Vector3.Dot(avatar.RightShoulderPosition-tip0,-tilt)+offset+AvatarVisual.CueGripFromButt,.75f,2.4f);var tip=tip0-tilt*visual;avatar.SetCue(tip,tilt,length,bridge);avatar.Animate(1f/60,AvatarPose.Aiming,10,false);if(shape==Archetype.Square && x==0 && z==0 && deg==0 && offset==.2f && (i==60 || i==90 || i==119)) Debug.Log($"MOTION i={i} visual={visual} shoulder={shoulder.position.ToString("F4")} elbow={elbow.position.ToString("F4")} wrist={avatar.RightHand.position.ToString("F4")} tip={tip.ToString("F4")}");if(i>=60){if(i==60) elbowAnchor=elbow.position; elbowTravel=Mathf.Max(elbowTravel,Vector3.Distance(elbowAnchor,elbow.position)); if(i>60) elbowStep=Mathf.Max(elbowStep,Vector3.Distance(elbowPrevious,elbow.position)); elbowPrevious=elbow.position;var wrist=avatar.RightHand;min=Mathf.Min(min,Vector3.Dot((elbow.position-wrist.position).normalized,wrist.up));var channel=wrist.TransformPoint(AvatarVisual.FistChannelLocal*AvatarVisual.FistScale);maxCueError=Mathf.Max(maxCueError,Vector3.ProjectOnPlane(channel-tip,tilt).magnitude);if(i>60)maxStep=Mathf.Max(maxStep,Quaternion.Angle(previous,wrist.rotation));previous=wrist.rotation;}}
 if(shape==Archetype.Square) Debug.Log($"CASEDATA x={x} z={z} deg={deg} travel={elbowTravel:F6} step={elbowStep:F6} align={min:F6}"); worstAlign=Mathf.Min(worstAlign,min); worst=Mathf.Max(worst,elbowTravel); worstStep=Mathf.Max(worstStep,elbowStep); sum+=elbowTravel;count++;} Debug.Log($"GRIPSCAN shape={shape} offset={offset} mix={mix} mean={sum/count} worst={worst} worstStep={worstStep} worstAlign={worstAlign}");}
 finally {Object.DestroyImmediate(holder);}
 }} }
